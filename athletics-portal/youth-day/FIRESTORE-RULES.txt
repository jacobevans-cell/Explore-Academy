// Add this match block inside your existing Cloud Firestore rules service.
// Public visitors may CREATE reservation requests only. They cannot read other families' data.
// The configured athletics admin email may read/update/delete them.

match /youthDaySignups/{signupId} {
  allow create: if request.resource.data.keys().hasAll([
      'schoolYear','playerName','team','contactName','contactEmail','selections','status','source','submittedAt'
    ])
    && request.resource.data.schoolYear == '2026-27'
    && request.resource.data.team in ['JV Volleyball','Varsity Volleyball','Boys Volleyball']
    && request.resource.data.status == 'active'
    && request.resource.data.source == 'explore-athletics-youth-day'
    && request.resource.data.playerName is string
    && request.resource.data.playerName.size() > 0
    && request.resource.data.playerName.size() <= 100
    && request.resource.data.contactName is string
    && request.resource.data.contactName.size() > 0
    && request.resource.data.contactName.size() <= 120
    && request.resource.data.contactEmail is string
    && request.resource.data.contactEmail.size() > 3
    && request.resource.data.contactEmail.size() <= 180
    && request.resource.data.selections is list
    && request.resource.data.selections.size() >= 1
    && request.resource.data.selections.size() <= 3;

  allow read, update, delete: if request.auth != null
    && request.auth.token.email == 'jacobicusjax@gmail.com';
}
